✅ Anti-Slop (What You Ship)

function processUserInput(
  data: UserInput
): ProcessedData[] {
  return data
    .filter((item): item is ValidInput => 
      item !== null && item.value !== undefined
    )
    .map(({ id, value }) => ({
      id,
      processedValue: transformValue(value),
      timestamp: Date.now()
    }));
}
